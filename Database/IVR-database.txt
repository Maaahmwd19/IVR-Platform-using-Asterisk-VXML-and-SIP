-- Creating the database
CREATE DATABASE ivr;

-- Connecting to the database
\c ivr

-- Creating tables and sequences
CREATE SEQUENCE users_user_id_seq;
CREATE TABLE users (
    user_id INTEGER PRIMARY KEY DEFAULT nextval('users_user_id_seq'),
    username VARCHAR(100) NOT NULL,
    msisdn VARCHAR(15) NOT NULL,
    balance NUMERIC(10,2) NOT NULL
);

CREATE SEQUENCE vxml_files_vxml_id_seq;
CREATE TABLE vxml_files (
    vxml_id INTEGER PRIMARY KEY DEFAULT nextval('vxml_files_vxml_id_seq'),
    file_name VARCHAR(100) NOT NULL,
    file_path VARCHAR(255) NOT NULL
);

CREATE SEQUENCE service_service_id_seq;
CREATE TABLE service (
    service_id INTEGER PRIMARY KEY DEFAULT nextval('service_service_id_seq'),
    service_name VARCHAR(100) NOT NULL,
    service_type VARCHAR(50) NOT NULL,
    quota INTEGER,
    service_fees NUMERIC(10,2) NOT NULL,
    vxml_id INTEGER NOT NULL,
    CONSTRAINT service_vxml_id_fkey FOREIGN KEY (vxml_id) REFERENCES vxml_files(vxml_id) ON DELETE RESTRICT
);

CREATE TABLE user_service (
    user_id INTEGER NOT NULL,
    service_id INTEGER NOT NULL,
    activation_status VARCHAR(8) NOT NULL,
    CONSTRAINT user_service_pkey PRIMARY KEY (user_id, service_id),
    CONSTRAINT user_service_user_id_fkey FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    CONSTRAINT user_service_service_id_fkey FOREIGN KEY (service_id) REFERENCES service(service_id) ON DELETE CASCADE,
    CONSTRAINT user_service_activation_status_check CHECK (activation_status IN ('Active', 'InActive'))
);

CREATE SEQUENCE sound_files_sound_id_seq;
CREATE TABLE sound_files (
    sound_id INTEGER PRIMARY KEY DEFAULT nextval('sound_files_sound_id_seq'),
    file_name VARCHAR(255) NOT NULL,
    file_path VARCHAR(255) NOT NULL,
    sound_vxml_name VARCHAR(255) NOT NULL,
    CONSTRAINT sound_files_sound_vxml_name_key UNIQUE (sound_vxml_name)
);

CREATE TABLE generated_sounds (
    sound_id SERIAL PRIMARY KEY,
    file_name VARCHAR(255) NOT NULL,
    file_path VARCHAR(255) NOT NULL,
    sound_vxml_name VARCHAR(255) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
